cmake_minimum_required(VERSION 3.20)
project(StudioPawnCompiler LANGUAGES C)
set(PAWN_SOURCE "" CACHE PATH "Pinned gf-pawncc source")
if(NOT EXISTS "${PAWN_SOURCE}/compiler/sc1.c")
  message(FATAL_ERROR "Set PAWN_SOURCE to the pinned gf-pawncc source directory")
endif()
set(sources sc1.c sc2.c sc3.c sc4.c sc5.c sc6.c sc7.c scexpand.c sci18n.c
    sclist.c scmemfil.c scstate.c scvars.c lstring.c memfile.c)
list(TRANSFORM sources PREPEND "${PAWN_SOURCE}/compiler/")
add_executable(gf-pawncc ${sources})
target_compile_definitions(gf-pawncc PRIVATE PAWN_CELL_SIZE=32)
include(CheckIncludeFile)
foreach(header unistd.h inttypes.h stdint.h alloca.h)
  string(TOUPPER "${header}" define)
  string(REPLACE "." "_" define "${define}")
  check_include_file("${header}" "HAVE_${define}")
  if(HAVE_${define})
    target_compile_definitions(gf-pawncc PRIVATE "HAVE_${define}")
  endif()
endforeach()
if(WIN32)
  target_compile_definitions(gf-pawncc PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_include_directories(gf-pawncc PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}/support")
  target_compile_definitions(gf-pawncc PRIVATE ENABLE_BINRELOC _I32_MAX=INT32_MAX _I32_MIN=INT32_MIN)
  target_compile_options(gf-pawncc PRIVATE -include unistd.h)
  target_sources(gf-pawncc PRIVATE support/binreloc.c)
  find_package(Threads REQUIRED)
  target_link_libraries(gf-pawncc PRIVATE Threads::Threads m)
endif()
set_target_properties(gf-pawncc PROPERTIES RUNTIME_OUTPUT_DIRECTORY "${CMAKE_BINARY_DIR}/bin")
foreach(config DEBUG RELEASE RELWITHDEBINFO MINSIZEREL)
  set_target_properties(gf-pawncc PROPERTIES RUNTIME_OUTPUT_DIRECTORY_${config} "${CMAKE_BINARY_DIR}/bin")
endforeach()
